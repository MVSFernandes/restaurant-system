select
  namespace.nspname as schema_name,
  table_row.relname as table_name,
  table_row.relrowsecurity as rls_enabled,
  table_row.relforcerowsecurity as rls_forced,
  count(policy.policyname) over (
    partition by table_row.oid
  ) as policy_count,
  policy.policyname as policy_name,
  policy.permissive,
  policy.roles,
  policy.cmd as command,
  policy.qual as using_expression,
  policy.with_check as check_expression
from pg_class as table_row
join pg_namespace as namespace
  on namespace.oid = table_row.relnamespace
left join pg_policies as policy
  on policy.schemaname = namespace.nspname
 and policy.tablename = table_row.relname
where namespace.nspname = 'public'
  and table_row.relkind in ('r', 'p')
order by
  table_row.relname,
  policy.policyname nulls first;
