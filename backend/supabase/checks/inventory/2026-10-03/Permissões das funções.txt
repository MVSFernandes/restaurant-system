with target_roles(role_name) as (
  values
    ('anon'),
    ('authenticated'),
    ('service_role')
),
public_routines as (
  select
    procedure_row.*,
    namespace.nspname as schema_name
  from pg_proc as procedure_row
  join pg_namespace as namespace
    on namespace.oid = procedure_row.pronamespace
  where namespace.nspname = 'public'
    and procedure_row.prokind in ('f', 'p')
)
select
  format(
    '%I.%I(%s)',
    routine.schema_name,
    routine.proname,
    pg_get_function_identity_arguments(routine.oid)
  ) as signature,
  target_role.role_name,
  database_role.oid is not null as role_exists,
  routine.proowner = database_role.oid as is_owner,
  coalesce(acl.direct_execute, false) as execute_granted_directly,
  coalesce(acl.public_execute, false) as execute_granted_via_public,
  case
    when database_role.oid is null then null
    else has_function_privilege(database_role.oid, routine.oid, 'EXECUTE')
  end as effective_execute,
  coalesce(acl.direct_grant_option, false) as direct_grant_option,
  routine.proacl is not null as acl_was_customized,
  case
    when database_role.oid is null then 'ROLE_MISSING'
    when coalesce(acl.direct_execute, false) then 'GRANTED_DIRECTLY'
    when coalesce(acl.public_execute, false) then 'GRANTED_VIA_PUBLIC'
    when has_function_privilege(database_role.oid, routine.oid, 'EXECUTE')
      then 'AVAILABLE_VIA_OWNERSHIP_OR_ROLE_MEMBERSHIP'
    else 'REVOKED_OR_NOT_GRANTED'
  end as execute_state
from public_routines as routine
cross join target_roles as target_role
left join pg_roles as database_role
  on database_role.rolname = target_role.role_name
left join lateral (
  select
    coalesce(
      bool_or(entry.privilege_type = 'EXECUTE')
        filter (where entry.grantee = database_role.oid),
      false
    ) as direct_execute,
    coalesce(
      bool_or(entry.privilege_type = 'EXECUTE')
        filter (where entry.grantee = 0),
      false
    ) as public_execute,
    coalesce(
      bool_or(entry.is_grantable)
        filter (
          where entry.grantee = database_role.oid
            and entry.privilege_type = 'EXECUTE'
        ),
      false
    ) as direct_grant_option
  from aclexplode(
    coalesce(
      routine.proacl,
      acldefault('f', routine.proowner)
    )
  ) as entry
) as acl on true
order by
  routine.proname,
  pg_get_function_identity_arguments(routine.oid),
  target_role.role_name;
