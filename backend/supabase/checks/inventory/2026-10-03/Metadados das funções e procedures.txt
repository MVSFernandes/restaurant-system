select
  namespace.nspname as schema_name,
  procedure_row.proname as routine_name,
  format(
    '%I.%I(%s)',
    namespace.nspname,
    procedure_row.proname,
    pg_get_function_identity_arguments(procedure_row.oid)
  ) as signature,
  case procedure_row.prokind
    when 'f' then 'FUNCTION'
    when 'p' then 'PROCEDURE'
  end as routine_kind,
  case
    when procedure_row.prokind = 'f'
      then pg_get_function_result(procedure_row.oid)
    else null
  end as result_type,
  language.lanname as language,
  owner_role.rolname as owner_name,
  procedure_row.prosecdef as security_definer,
  case procedure_row.provolatile
    when 'i' then 'IMMUTABLE'
    when 's' then 'STABLE'
    when 'v' then 'VOLATILE'
  end as volatility,
  case procedure_row.proparallel
    when 's' then 'SAFE'
    when 'r' then 'RESTRICTED'
    when 'u' then 'UNSAFE'
  end as parallel_safety,
  procedure_row.proconfig as runtime_settings
from pg_proc as procedure_row
join pg_namespace as namespace
  on namespace.oid = procedure_row.pronamespace
join pg_language as language
  on language.oid = procedure_row.prolang
join pg_roles as owner_role
  on owner_role.oid = procedure_row.proowner
where namespace.nspname = 'public'
  and procedure_row.prokind in ('f', 'p')
order by
  procedure_row.proname,
  pg_get_function_identity_arguments(procedure_row.oid);
