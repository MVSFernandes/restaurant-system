select
  procedure_row.proname as routine_name,
  format(
    '%I.%I(%s)',
    namespace.nspname,
    procedure_row.proname,
    pg_get_function_identity_arguments(procedure_row.oid)
  ) as signature,
  pg_get_functiondef(procedure_row.oid) as routine_definition
from pg_proc as procedure_row
join pg_namespace as namespace
  on namespace.oid = procedure_row.pronamespace
where namespace.nspname = 'public'
  and procedure_row.prokind in ('f', 'p')
  and not exists (
    select 1
    from pg_depend as dependency
    join pg_extension as extension
      on extension.oid = dependency.refobjid
    where dependency.classid = 'pg_proc'::regclass
      and dependency.objid = procedure_row.oid
      and dependency.deptype = 'e'
  )
order by
  procedure_row.proname,
  pg_get_function_identity_arguments(procedure_row.oid);
