select
  indexes.tablename as table_name,
  indexes.indexname as index_name,
  case constraint_row.contype
    when 'p' then 'PRIMARY_KEY_CONSTRAINT'
    when 'u' then 'UNIQUE_CONSTRAINT'
    when 'x' then 'EXCLUDE_CONSTRAINT'
    else 'STANDALONE_INDEX'
  end as index_origin,
  index_row.indisunique as is_unique,
  index_row.indisvalid as is_valid,
  indexes.indexdef
from pg_indexes as indexes
join pg_class as index_class
  on index_class.relname = indexes.indexname
join pg_namespace as index_namespace
  on index_namespace.oid = index_class.relnamespace
 and index_namespace.nspname = indexes.schemaname
join pg_index as index_row
  on index_row.indexrelid = index_class.oid
left join pg_constraint as constraint_row
  on constraint_row.conindid = index_class.oid
 and constraint_row.contype in ('p', 'u', 'x')
where indexes.schemaname = 'public'
    and indexes.tablename >= 'orders'
order by indexes.tablename, indexes.indexname;
