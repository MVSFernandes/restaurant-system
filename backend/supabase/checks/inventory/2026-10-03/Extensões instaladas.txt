select
  extension.extname as extension_name,
  extension.extversion as installed_version,
  namespace.nspname as installed_schema,
  extension.extrelocatable as is_relocatable
from pg_extension as extension
join pg_namespace as namespace
  on namespace.oid = extension.extnamespace
order by extension.extname;

| pg_trgm            | 1.6               | public           | true           |
| pgcrypto           | 1.3               | extensions       | true           |

| supabase_vault     | 0.3.1             | vault            | false          |
