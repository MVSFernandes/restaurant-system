select
  namespace.nspname as schema_name,
  table_row.relname as table_name,
  attribute.attnum as ordinal_position,
  attribute.attname as column_name,
  format_type(attribute.atttypid, attribute.atttypmod) as data_type,
  not attribute.attnotnull as is_nullable,
  pg_get_expr(default_row.adbin, default_row.adrelid) as column_default,
  case attribute.attidentity
    when 'a' then 'GENERATED ALWAYS'
    when 'd' then 'GENERATED BY DEFAULT'
    else null
  end as identity_kind,
  case attribute.attgenerated
    when 's' then 'STORED'
    when 'v' then 'VIRTUAL'
    else null
  end as generated_kind
from pg_class as table_row
join pg_namespace as namespace
  on namespace.oid = table_row.relnamespace
join pg_attribute as attribute
  on attribute.attrelid = table_row.oid
left join pg_attrdef as default_row
  on default_row.adrelid = table_row.oid
 and default_row.adnum = attribute.attnum
where namespace.nspname = 'public'
  and table_row.relkind in ('r', 'p')
  and attribute.attnum > 0
  and not attribute.attisdropped
    and table_row.relname >= 'products'
order by table_row.relname, attribute.attnum;
